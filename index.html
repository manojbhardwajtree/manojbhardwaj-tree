let currentRole = null; const FAMILY_STORAGE_KEY = "mb_family_ledger_clean_v1"; const FAMILY_CREDS_KEY = "mb_family_creds_clean_v1"; const DEFAULT_USERS = { admin: { id: "Admin", pass: "admin123", role: "ADMIN" }, family: { id: "Family", pass: "family123", role: "FAMILY" } }; function loadCreds() { try { const raw = localStorage.getItem(FAMILY_CREDS_KEY); return raw ? JSON.parse(raw) : DEFAULT_USERS; } catch (e) { return DEFAULT_USERS; } } function loadRecords() { try { const raw = localStorage.getItem(FAMILY_STORAGE_KEY); return raw ? JSON.parse(raw) : new Array(); } catch (e) { return new Array(); } } function saveRecords(list) { localStorage.setItem(FAMILY_STORAGE_KEY, JSON.stringify(list)); } function findField(idName, labelKeyword) { const byId = document.getElementById(idName); if (byId) return byId; const portal = document.getElementById("mainPortal"); if (!portal) return null; const labels = portal.querySelectorAll("label"); for (let i = 0; i < labels.length; i++) { const txt = (labels[i].textContent || "").toLowerCase(); if (txt.indexOf(labelKeyword.toLowerCase()) !== -1) { const next = labels[i].nextElementSibling; if (next && (next.tagName === "INPUT" || next.tagName === "SELECT" || next.tagName === "TEXTAREA")) { return next; } if (labels[i].parentElement) { const inner = labels[i].parentElement.querySelector("input, select, textarea"); if (inner) return inner; } } } return null; } function getAllFamilyFields() { return { id: findField("mId", "member id"), order: findField("mOrder", "sibling birth order"), gen: findField("mGen", "generation"), branch: findField("mBranch", "lineage branch"), first: findField("mFirst", "first name"), last: findField("mLast", "last name"), father: findField("mFather", "father id"), mother: findField("mMother", "mother id"), siblings: findField("mSiblings", "siblings"), status: findField("mStatus", "status"), dob: findField("mDob", "date of birth"), tob: findField("mTob", "time of birth"), pob: findField("mPob", "birth place"), gotra: findField("mGotra", "gotra"), email: findField("mEmail", "email"), contact: findField("mContact", "contact"), locked: findField("mLocked", "admin verified") }; } window.addEventListener("DOMContentLoaded", function() { const fSec = document.getElementById("familySection"); if (!fSec) return; const btns = fSec.querySelectorAll("button"); btns.forEach(function(b) { const label = (b.textContent || "").trim(); if (label.indexOf("Save Record") !== -1 && !b.getAttribute("onclick")) { b.type = "button"; b.addEventListener("click", saveFamilyRecord); } else if (label.indexOf("Clear") !== -1 && !b.getAttribute("onclick")) { b.type = "button"; b.addEventListener("click", clearFamilyForm); } else if (label.indexOf("Sign Out of Family Portal") !== -1 && !b.getAttribute("onclick")) { b.type = "button"; b.addEventListener("click", signOut); } }); const pwdInput = document.getElementById("pwd"); if (pwdInput) { pwdInput.addEventListener("keypress", function(e) { if (e.key === "Enter") unlock(); }); } }); function unlock() { const uidVal = (document.getElementById("uid").value || "").trim().toLowerCase(); const pwdVal = (document.getElementById("pwd").value || "").trim(); const errBox = document.getElementById("err"); const users = loadCreds(); if (uidVal === users.admin.id.toLowerCase() && pwdVal === users.admin.pass) { currentRole = "ADMIN"; } else if (uidVal === users.family.id.toLowerCase() && pwdVal === users.family.pass) { currentRole = "FAMILY"; } else { if (errBox) errBox.style.display = "block"; return; } if (errBox) errBox.style.display = "none"; document.getElementById("loginBox").style.display = "none"; document.getElementById("mainPortal").style.display = "block"; const badge = document.getElementById("roleBadge"); if (badge) { badge.textContent = currentRole; badge.className = currentRole === "ADMIN" ? "badge-admin" : "badge-family"; } const pushBtn = document.getElementById("btnPushToFamily"); const reportBox = document.getElementById("kundliReport"); if (pushBtn && reportBox && reportBox.style.display === "block") { pushBtn.style.display = "inline-block"; } renderFamilyTable(); } function signOut() { currentRole = null; const pwdInput = document.getElementById("pwd"); if (pwdInput) pwdInput.value = ""; document.getElementById("mainPortal").style.display = "none"; document.getElementById("loginBox").style.display = "block"; const pushBtn = document.getElementById("btnPushToFamily"); if (pushBtn) pushBtn.style.display = "none"; } function changePass() { if (!currentRole) return; const nextPass = prompt("Enter new password for " + currentRole + " (minimum 4 characters):"); if (!nextPass || nextPass.trim().length < 4) { alert("Password must be at least 4 characters."); return; } const users = loadCreds(); if (currentRole === "ADMIN") users.admin.pass = nextPass.trim(); else users.family.pass = nextPass.trim(); localStorage.setItem(FAMILY_CREDS_KEY, JSON.stringify(users)); alert("✅ Password updated for " + currentRole + "."); } function saveFamilyRecord() { const f = getAllFamilyFields(); const first = f.first ? f.first.value.trim() : ""; if (!first) { alert("Please enter First Name."); return; } const records = loadRecords(); const recId = (f.id && f.id.value.trim()) ? f.id.value.trim() : ("M_" + (records.length + 1)); const idx = records.findIndex(function(r) { return r.id === recId; }); if (idx >= 0 && records[idx].locked && currentRole !== "ADMIN") { alert("🛡️ This record is Admin Verified & Locked. Only Admin can edit it."); return; } const entry = { id: recId, order: f.order ? f.order.value.trim() : "1", gen: f.gen ? f.gen.value : "G1_00", branch: f.branch ? f.branch.value : "Paternal Lineage", first: first, last: f.last ? f.last.value.trim() : "", father: f.father ? f.father.value.trim() : "", mother: f.mother ? f.mother.value.trim() : "", siblings: f.siblings ? f.siblings.value.trim() : "", status: f.status ? f.status.value : "Alive", dob: f.dob ? f.dob.value : "", tob: f.tob ? f.tob.value : "", pob: f.pob ? f.pob.value.trim() : "", gotra: f.gotra ? f.gotra.value.trim() : "", email: f.email ? f.email.value.trim() : "", contact: f.contact ? f.contact.value.trim() : "", locked: (currentRole === "ADMIN" && f.locked) ? Boolean(f.locked.checked) : (idx >= 0 ? records[idx].locked : false) }; if (idx >= 0) records[idx] = entry; else records.push(entry); saveRecords(records); clearFamilyForm(); renderFamilyTable(); alert("✅ Saved record for " + entry.first + " (" + entry.id + ")"); } function clearFamilyForm() { const f = getAllFamilyFields(); Object.keys(f).forEach(function(k) { const el = f[k]; if (!el) return; if (el.type === "checkbox") el.checked = false; else if (el.tagName === "SELECT") el.selectedIndex = 0; else el.value = ""; }); } function renderFamilyTable() { const tbody = document.querySelector("#mainPortal tbody"); if (!tbody) return; const records = loadRecords(); tbody.innerHTML = ""; records.forEach(function(r) { const lockTag = r.locked ? " 🛡️" : ""; const placeInfo = new Array(r.pob, r.tob, r.status, r.gotra).filter(Boolean).join(" | "); const contactInfo = new Array(r.contact, r.email).filter(Boolean).join(" / "); const tr = document.createElement("tr"); tr.innerHTML = "" + r.id + "" + lockTag + "" + "" + (r.order || "-") + "" + "" + r.first + " " + r.last + "" + "" + r.branch + " (" + r.gen + ")" + "" + (r.father || "-") + "" + "" + (r.dob || "-") + "" + "" + (placeInfo || "-") + "" + "" + (contactInfo || "-") + "" + " " + ""; tbody.appendChild(tr); }); } function editFamilyRecord(id) { const records = loadRecords(); const r = records.find(function(item) { return item.id === id; }); if (!r) return; if (r.locked && currentRole !== "ADMIN") { alert("🛡️ This record is Admin Verified & Locked."); return; } const f = getAllFamilyFields(); if (f.id) f.id.value = r.id; if (f.order) f.order.value = r.order; if (f.gen) f.gen.value = r.gen; if (f.branch) f.branch.value = r.branch; if (f.first) f.first.value = r.first; if (f.last) f.last.value = r.last; if (f.father) f.father.value = r.father; if (f.mother) f.mother.value = r.mother; if (f.siblings) f.siblings.value = r.siblings; if (f.status) f.status.value = r.status; if (f.dob) f.dob.value = r.dob; if (f.tob) f.tob.value = r.tob; if (f.pob) f.pob.value = r.pob; if (f.gotra) f.gotra.value = r.gotra; if (f.email) f.email.value = r.email; if (f.contact) f.contact.value = r.contact; if (f.locked) f.locked.checked = Boolean(r.locked); } function deleteFamilyRecord(id) { const records = loadRecords(); const r = records.find(function(item) { return item.id === id; }); if (!r) return; if (r.locked && currentRole !== "ADMIN") { alert("🛡️ Only Admin can delete a locked record."); return; } if (!confirm("Delete record " + id + " (" + r.first + ")?")) return; saveRecords(records.filter(function(item) { return item.id !== id; })); renderFamilyTable(); } function exportCSV() { const records = loadRecords(); if (!records.length) { alert("No family records to export yet."); return; } const headers = new Array("ID", "Order", "Generation", "Branch", "First Name", "Last Name", "Father ID", "Mother ID", "Siblings", "Status", "DOB", "TOB", "Place/Notes", "Gotra", "Email", "Contact", "Locked"); const rows = new Array(headers.join(",")); records.forEach(function(r) { const vals = new Array(r.id, r.order, r.gen, r.branch, r.first, r.last, r.father, r.mother, r.siblings, r.status, r.dob, r.tob, r.pob, r.gotra, r.email, r.contact, r.locked ? "Yes" : "No"); rows.push(vals.map(function(v) { return '"' + String(v || "").replace(/"/g, '""') + '"'; }).join(",")); }); const blob = new Blob([rows.join("\n")], { type: "text/csv;charset=utf-8;" }); const link = document.createElement("a"); link.href = URL.createObjectURL(blob); link.download = "ManojBhardwaj_Family_Ledger.csv"; link.click(); }
